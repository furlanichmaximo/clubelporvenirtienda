  function signIn(user) {
    state.user = user;
    save();
    renderAccount();
    closeAll();
    notify(`¡Hola, ${firstName(user.name)}!`, 'Ingresaste a tu cuenta');
  }

  function submitAuth(e) {
    e.preventDefault();
    const form = e.target, reg = form.dataset.mode === 'register';
    const name = reg ? $('#a-name').value.trim() : '';
    const email = $('#a-mail').value.trim();
    const pass = $('#a-pass').value;
    form.querySelectorAll('input').forEach(i => (i.style.borderColor = ''));
    const bad = (el, msg) => { el.style.borderColor = 'var(--red)'; el.focus(); $('#authErr').textContent = msg; };
    if (reg && !name) return bad($('#a-name'), 'Ingresá tu nombre.');
    if (!/^\S+@\S+\.\S+$/.test(email)) return bad($('#a-mail'), 'Ingresá un e-mail válido.');
    if (pass.length < 6) return bad($('#a-pass'), 'La contraseña debe tener al menos 6 caracteres.');
    const btn = form.querySelector('[type="submit"]');
    btn.disabled = true;
    btn.innerHTML = '<span class="spinner"></span>';
    const local = email.split('@')[0].replace(/[._-]+/g, ' ');
    setTimeout(() => signIn({ name: name || local.charAt(0).toUpperCase() + local.slice(1), email, provider: 'email' }), 800);
  }

  function socialSignIn(btn) {
    const provider = btn.dataset.social;
    $$('.social-btn').forEach(b => (b.disabled = true));
    btn.innerHTML = `<span class="spinner"></span> Conectando con ${provider}...`;
    setTimeout(() => signIn({ name: 'Hincha del Porve', email: `hincha.porve@${provider === 'Google' ? 'gmail.com' : 'facebook.com'}`, provider }), 1100);
  }

  function notify(title, sub) {
    const el = $('#toast');
    el.innerHTML = `<div><b>${esc(title)}</b><span>${esc(sub)}</span></div>`;
    el.classList.add('show');
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => el.classList.remove('show'), 2800);
  }

  // ---------- Toast ----------

  // ---------- Events ----------
  document.addEventListener('click', e => {
    if (!e.target.closest('#account')) toggleAcctMenu(false);
    const t = e.target.closest('button, a, .overlay');
    if (!t) return;

    if (t.id === 'acctBtn') return state.user ? toggleAcctMenu(!$('#acctMenu').classList.contains('open')) : openAuth('login');
    if (t.hasAttribute('data-logout')) { toggleAcctMenu(false); state.user = null; save(); renderAccount(); return notify('Cerraste sesión', '¡Te esperamos de vuelta!'); }
    if (t.dataset.authMode) return openAuth(t.dataset.authMode);
    if (t.dataset.social) return socialSignIn(t);
    if (t.hasAttribute('data-mp-cancel')) return openCheckout();

    if (t.dataset.nav) {
      e.preventDefault();
      if (openLayer) closeAll();
    if (t.dataset.quick) return openQuickView(t.dataset.quick);
    if (t.dataset.qty) return changeQty(+t.dataset.qty, +t.dataset.d);
    if (t.dataset.rm) { state.cart.splice(+t.dataset.rm, 1); save(); renderCart(); return; }
    if (t.hasAttribute('data-open-cart') || t.id === 'cartBtn') { $('#toast').classList.remove('show'); return openDrawer(); }
    if (t.hasAttribute('data-open-cart') || t.id === 'cartBtn') { toggleAcctMenu(false); $('#toast').classList.remove('show'); return openDrawer(); }
    if (t.id === 'checkoutBtn') return openCheckout();
    if (t.hasAttribute('data-back')) return openDrawer();
    if (t.hasAttribute('data-close') || t.classList.contains('overlay')) return closeAll();

  // Click outside a modal box closes it
  $$('.modal').forEach(m => m.addEventListener('click', e => { if (e.target === m) closeAll(); }));
  document.addEventListener('keydown', e => { if (e.key === 'Escape') { closeAll(); closeMobileNav(); } });
  document.addEventListener('submit', e => { if (e.target.id === 'coForm') submitCheckout(e); });
  document.addEventListener('keydown', e => { if (e.key === 'Escape') { closeAll(); closeMobileNav(); toggleAcctMenu(false); } });
  document.addEventListener('submit', e => {
    if (e.target.id === 'coForm') submitCheckout(e);
    if (e.target.id === 'authForm') submitAuth(e);
  });

  $$('input[name="delivery"]').forEach(r => r.addEventListener('change', () => { state.delivery = r.value; save(); renderCart(); }));
  $$('input[name="payment"]').forEach(r => r.addEventListener('change', () => { state.payment = r.value; save(); }));
  $('#socioToggle').addEventListener('change', e => { state.socio = e.target.checked; save(); renderCart(); });

  let searchTimer;

  renderGrid();
  renderCart();
  renderAccount();
})();
